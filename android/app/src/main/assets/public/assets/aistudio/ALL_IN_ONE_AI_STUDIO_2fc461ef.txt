GOAL
Build a WEB APP (React + TypeScript) where users sign up, create or join
password-protected rooms, opt in to share live location while the page is open,
and see room members on a map with distance, "last updated", and accuracy.
Backend is Supabase (Postgres + PostGIS + RLS). Privacy is enforced in the
database, not in the UI.

IMPORTANT: The code in PART 3 is React Native / Expo. AI Studio builds web apps,
so treat PART 3 as REFERENCE LOGIC and port it. Do not paste it unchanged.

PORTING MAP (React Native reference -> web)
- expo-router screens            -> React Router routes (or simple tab state)
- <View>, <Text>, <Pressable>    -> div, span, button with CSS
- <TextInput>                    -> input
- Alert.alert                    -> inline message or toast (no window.alert spam)
- AsyncStorage (session)         -> Supabase default localStorage (remove custom storage)
- expo-location watchPosition    -> navigator.geolocation.watchPosition
- getCurrentPositionAsync        -> navigator.geolocation.getCurrentPosition
- react-native-maps              -> Leaflet or MapLibre GL in a single MapSurface component
- AppState foreground refresh    -> document visibilitychange event
- app.json permission text       -> show an explanation BEFORE calling geolocation
- "react-native-url-polyfill"    -> not needed on web

MAP NOTES
- Keep ALL map-library code inside one MapSurface component so the provider can change.
- Standard map: OpenStreetMap-based tiles are fine for light use, but check the tile
  provider's usage policy. Satellite needs a LICENSED provider (e.g. Mapbox or MapTiler
  with your own key). Do not scrape Google or Esri tiles without checking their terms.
- Markers show initials for now (no avatars).

LOCATION NOTES
- Browsers share location only on HTTPS and only while the page is open. There is no
  background tracking on the web. Say so in the UI.
- If the AI Studio preview blocks location, the preview iframe may need the geolocation
  permission declared (AI Studio apps use a metadata.json for frame permissions).
  Check current AI Studio docs for the exact setting. If it still fails, deploy the app
  to a normal HTTPS host to test.
- The browser Geolocation accuracy value is real; display it. Never invent accuracy.

HARD RULES
1. Do NOT invent APIs, package names, or SQL. If unsure, say so and ask.
2. Do NOT put the service_role key anywhere. Anon key only.
3. Do NOT fake users, locations, messages, or statistics. Start with an empty app.
4. Server timestamps decide "Live". Never trust the browser clock for that.
   "Live" only if the server's updated_at is under 2 minutes old.
5. Distances come from the room_member_locations RPC (PostGIS), not from pixels.
6. Sharing is OFF by default and expires automatically (default 1 hour).
7. join_room returns the text status 'ok' | 'invalid' | 'rate_limited'. Show ONE message
   for wrong ID and wrong password: "Invalid room ID or password".
8. Every screen needs loading, empty, and error-with-retry states.
9. Work in phases. After each phase list what changed and what you could not verify.
   If something fails twice, stop and show me the exact error.

Phase B: Rooms (list my rooms, create, join).
Phase C: Map (MapSurface, members with distance/freshness/accuracy, sharing toggle,

Phase D: Final report: files changed, checks run, what is unverified, how to test with

######## FILE: 001_mvp_schema.sql ########
-- MVP schema: profiles, rooms, members, live location, chat, block/report
-- Supabase (Postgres + PostGIS). Run in the SQL editor or as a migration.

create extension if not exists postgis;
create extension if not exists pgcrypto with schema extensions;

-- ========== TABLES ==========

create table profiles (
  id          uuid primary key references auth.users on delete cascade,
  username    text unique not null check (username ~ '^[a-z0-9_]{3,24}$'),
  full_name   text not null check (char_length(full_name) between 1 and 80),
  about       text check (char_length(about) <= 200),
  avatar_url  text,
  created_at  timestamptz not null default now()
);

create table rooms (
  id          uuid primary key default gen_random_uuid(),
  name        text not null check (char_length(name) between 1 and 60),
  description text check (char_length(description) <= 300),
  icon_url    text,
  owner_id    uuid not null references profiles on delete cascade,
  created_at  timestamptz not null default now()
);

-- Password hashes live apart from rooms. RLS is on with NO policies,
-- so clients can never read them. Only security-definer functions can.
create table room_secrets (
  room_id       uuid primary key references rooms on delete cascade,
  password_hash text not null
);

create type room_role as enum ('owner', 'member');

create table room_members (
  room_id           uuid references rooms on delete cascade,
  user_id           uuid references profiles on delete cascade,
  role              room_role not null default 'member',
  sharing_enabled   boolean not null default false,  -- off by default
  share_expires_at  timestamptz,                      -- null = until turned off
  joined_at         timestamptz not null default now(),
  primary key (room_id, user_id)
);
create index on room_members (user_id);

-- Latest position only. No location history is kept (data minimization).
create table user_locations (
  user_id     uuid primary key references profiles on delete cascade,
  geog        geography(Point, 4326) not null,
  accuracy_m  real check (accuracy_m >= 0),
  updated_at  timestamptz not null default now()
);
create index on user_locations using gist (geog);

create table messages (
  id          bigint generated always as identity primary key,
  room_id     uuid not null references rooms on delete cascade,
  user_id     uuid not null references profiles on delete cascade,
  body        text not null check (char_length(body) between 1 and 4000),
  reply_to    bigint references messages on delete set null,
  created_at  timestamptz not null default now()
);
create index on messages (room_id, created_at desc);

create table blocks (
  blocker_id  uuid references profiles on delete cascade,
  blocked_id  uuid references profiles on delete cascade,
  created_at  timestamptz not null default now(),
  primary key (blocker_id, blocked_id),
  check (blocker_id <> blocked_id)
);

create table reports (
  id           bigint generated always as identity primary key,
  reporter_id  uuid not null references profiles on delete cascade,
  target_user  uuid references profiles on delete set null,
  message_id   bigint references messages on delete set null,
  reason       text not null check (char_length(reason) between 1 and 500),
  created_at   timestamptz not null default now()
);

-- ========== HELPER FUNCTIONS ==========
-- Security definer avoids RLS recursion on room_members.

create function is_room_member(p_room uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from room_members
                 where room_id = p_room and user_id = auth.uid());
$$;

create function is_room_owner(p_room uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from room_members
                 where room_id = p_room and user_id = auth.uid() and role = 'owner');
$$;

create function shares_room_with(p_other uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from room_members a
    join room_members b on a.room_id = b.room_id
    where a.user_id = auth.uid() and b.user_id = p_other
  );
$$;

create function is_blocked_between(p_other uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from blocks
    where (blocker_id = auth.uid() and blocked_id = p_other)
       or (blocker_id = p_other and blocked_id = auth.uid())
  );
$$;

-- The core privacy rule: viewer may see target's location only if
-- they share a room where the target has sharing ON and not expired,
-- and neither has blocked the other.
create function can_see_location(p_target uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select p_target = auth.uid()
  or (
    exists (
      select 1 from room_members me
      join room_members them on me.room_id = them.room_id
      where me.user_id = auth.uid()
        and them.user_id = p_target
        and them.sharing_enabled
        and (them.share_expires_at is null or them.share_expires_at > now())
    )
    and not is_blocked_between(p_target)
  );
$$;

-- ========== RPCs (the only way to create/join rooms) ==========

create function create_room(p_name text, p_description text, p_password text)
returns uuid language plpgsql security definer set search_path = public, extensions as $$
declare v_id uuid;
begin
  if char_length(p_password) < 8 then
    raise exception 'Password must be at least 8 characters';
  end if;
  insert into rooms (name, description, owner_id)
    values (p_name, p_description, auth.uid()) returning id into v_id;
  insert into room_secrets (room_id, password_hash)
    values (v_id, crypt(p_password, gen_salt('bf', 10)));
  insert into room_members (room_id, user_id, role)
    values (v_id, auth.uid(), 'owner');
  return v_id;
end $$;

-- TODO before launch: add attempt rate limiting (table of failed joins
-- per user/room, lock after N tries) to stop password brute-forcing.
create function join_room(p_room_id uuid, p_password text)
returns void language plpgsql security definer set search_path = public, extensions as $$
declare v_hash text;
begin
  select password_hash into v_hash from room_secrets where room_id = p_room_id;
  -- Same error for "no such room" and "wrong password" (no room enumeration)
  if v_hash is null or crypt(p_password, v_hash) <> v_hash then
    raise exception 'Invalid room ID or password';
  end if;
  insert into room_members (room_id, user_id)
    values (p_room_id, auth.uid()) on conflict do nothing;
end $$;

-- Lock RPCs to signed-in users only
revoke execute on function create_room, join_room from public, anon;
grant  execute on function create_room, join_room to authenticated;

-- Nearby members' locations with distance in meters (computed server-side)
create function room_member_locations(p_room uuid)
returns table (user_id uuid, lat double precision, lng double precision,
               accuracy_m real, updated_at timestamptz, distance_m double precision)
language sql stable security definer set search_path = public as $$
  select l.user_id,
         st_y(l.geog::geometry), st_x(l.geog::geometry),
         l.accuracy_m, l.updated_at,
         st_distance(l.geog, mine.geog)
  from user_locations l
  join room_members rm on rm.user_id = l.user_id and rm.room_id = p_room
  left join user_locations mine on mine.user_id = auth.uid()
  where is_room_member(p_room)
    and l.user_id <> auth.uid()
    and can_see_location(l.user_id)
    and rm.sharing_enabled
    and (rm.share_expires_at is null or rm.share_expires_at > now());
$$;

alter table profiles       enable row level security;
alter table rooms          enable row level security;
alter table room_secrets   enable row level security;  -- no policies = no access
alter table room_members   enable row level security;
alter table user_locations enable row level security;
alter table messages       enable row level security;
alter table blocks         enable row level security;
alter table reports        enable row level security;

create policy profiles_select on profiles for select to authenticated

create policy profiles_insert on profiles for insert to authenticated
  with check (id = auth.uid());
create policy profiles_update on profiles for update to authenticated
  using (id = auth.uid()) with check (id = auth.uid());

-- rooms: members read; owner edits. Creation goes through create_room().
create policy rooms_select on rooms for select to authenticated
  using (is_room_member(id));
create policy rooms_update on rooms for update to authenticated
  using (is_room_owner(id)) with check (is_room_owner(id));
create policy rooms_delete on rooms for delete to authenticated
  using (is_room_owner(id));

-- room_members: members see co-members; you may edit only YOUR sharing
-- settings (column-level grant below stops role changes); you may leave.
create policy members_select on room_members for select to authenticated
  using (is_room_member(room_id));
create policy members_update_self on room_members for update to authenticated
  using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy members_leave on room_members for delete to authenticated
  using (user_id = auth.uid() and role <> 'owner');

revoke update on room_members from authenticated;
grant  update (sharing_enabled, share_expires_at) on room_members to authenticated;

-- user_locations: you write your own; reads go through can_see_location()
create policy loc_select on user_locations for select to authenticated
  using (can_see_location(user_id));
create policy loc_insert on user_locations for insert to authenticated
  with check (user_id = auth.uid());
create policy loc_update on user_locations for update to authenticated
  using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy loc_delete on user_locations for delete to authenticated
  using (user_id = auth.uid());

-- messages
create policy msg_select on messages for select to authenticated
  using (is_room_member(room_id) and not is_blocked_between(user_id));
create policy msg_insert on messages for insert to authenticated
  with check (user_id = auth.uid() and is_room_member(room_id));
create policy msg_delete on messages for delete to authenticated

-- blocks / reports
create policy blocks_all on blocks for all to authenticated
  using (blocker_id = auth.uid()) with check (blocker_id = auth.uid());
create policy reports_insert on reports for insert to authenticated
  with check (reporter_id = auth.uid());

######## FILE: 004_join_rate_limit.sql ########
-- Rate limiting for join_room(). Run after 001_mvp_schema.sql.
--
-- WHY IT RETURNS A STATUS INSTEAD OF RAISING:
-- A raised exception rolls back the transaction, including the row that logs
-- the failed attempt, so the lockout would never trigger. join_room now returns
-- text: 'ok' | 'invalid' | 'rate_limited'. Raise only for unauthenticated calls.
--
-- THREAT MODEL: room IDs are random UUIDs, so an attacker needs a leaked ID
-- before guessing the password. These limits slow that guessing.

drop function if exists join_room(uuid, text);

create table join_attempts (
  id            bigint generated always as identity primary key,
  user_id       uuid not null references profiles on delete cascade,
  room_id       uuid not null,   -- no FK on purpose: also logs nonexistent rooms
  succeeded     boolean not null,
  attempted_at  timestamptz not null default now()
);
create index on join_attempts (user_id, attempted_at desc);
create index on join_attempts (room_id, attempted_at desc);

alter table join_attempts enable row level security;  -- no policies = clients can't touch it

create function join_room(p_room_id uuid, p_password text)
returns text language plpgsql security definer set search_path = public, extensions as $$
declare
  v_uid   uuid := auth.uid();
  v_hash  text;
  v_ok    boolean;
begin
  if v_uid is null then
    raise exception 'Not authenticated';
  end if;

  -- Limit 1: 10 failures per user across all rooms in 15 minutes
  if (select count(*) from join_attempts
      where user_id = v_uid and not succeeded
        and attempted_at > now() - interval '15 minutes') >= 10 then
    return 'rate_limited';
  end if;

  -- Limit 2: 5 failures per user per room in 15 minutes
  if (select count(*) from join_attempts
      where user_id = v_uid and room_id = p_room_id and not succeeded
        and attempted_at > now() - interval '15 minutes') >= 5 then
    return 'rate_limited';
  end if;

  -- Limit 3: 100 failures per room per hour from anyone (defends against
  -- many throwaway accounts). Trade-off: a determined attacker with a leaked
  -- room ID can briefly block legitimate joins to that room.
  if (select count(*) from join_attempts
      where room_id = p_room_id and not succeeded
        and attempted_at > now() - interval '1 hour') >= 100 then
    return 'rate_limited';
  end if;

  select password_hash into v_hash from room_secrets where room_id = p_room_id;

  if v_hash is null then
    -- Burn the same bcrypt time as a real check so response time
    -- doesn't reveal whether a room exists.
    perform crypt(p_password, gen_salt('bf', 10));
    v_ok := false;
  else
    v_ok := crypt(p_password, v_hash) = v_hash;
  end if;

  insert into join_attempts (user_id, room_id, succeeded)
    values (v_uid, p_room_id, v_ok);

  if not v_ok then
    return 'invalid';   -- same answer for wrong password and unknown room
  end if;

  insert into room_members (room_id, user_id)
    values (p_room_id, v_uid) on conflict do nothing;
  return 'ok';
end $$;

revoke execute on function join_room(uuid, text) from public, anon;
grant  execute on function join_room(uuid, text) to authenticated;

######## FILE: 005_profile_trigger_and_timestamps.sql ########
-- Two fixes the app needs. Run after 001.

-- 1. Create the profile row when someone signs up.
--    The app passes username and full_name as sign-up metadata. Doing this in a
--    trigger works even when email confirmation is on (no session at sign-up).
create function handle_new_user()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  insert into profiles (id, username, full_name)
  values (
    new.id,
    coalesce(nullif(new.raw_user_meta_data ->> 'username', ''), 'user_' || substr(new.id::text, 1, 8)),
    coalesce(nullif(new.raw_user_meta_data ->> 'full_name', ''), 'New user')
  );
  return new;
end $$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function handle_new_user();

-- 2. Location timestamps come from the SERVER clock, never the phone's.
--    Otherwise a wrong device clock could make a stale marker look "Live".
create function touch_location()
returns trigger language plpgsql as $$
begin
  new.updated_at := now();
  return new;
end $$;

create trigger user_locations_touch
  before insert or update on user_locations
  for each row execute function touch_location();
######## END FILE: 005_profile_trigger_and_timestamps.sql ########

-- Storage buckets and policies. Run after 001_mvp_schema.sql.
-- All buckets are PRIVATE. The app reads files through short-lived signed URLs,
-- so a leaked link expires and strangers can't browse.
--
-- Path conventions (policies depend on these):
--   avatars     : {user_id}/{filename}
--   room-icons  : {room_id}/{filename}
--   chat-media  : {room_id}/{filename}      (week 6+, optional for MVP)

-- ========== BUCKETS ==========
-- No SVG: SVG files can carry scripts. Raster images only.
insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types) values
  ('avatars',    'avatars',    false, 2 * 1024 * 1024,  array['image/jpeg','image/png','image/webp']),
  ('room-icons', 'room-icons', false, 2 * 1024 * 1024,  array['image/jpeg','image/png','image/webp']),
  ('chat-media', 'chat-media', false, 10 * 1024 * 1024, array['image/jpeg','image/png','image/webp'])
on conflict (id) do update
  set public = false,
      file_size_limit = excluded.file_size_limit,
      allowed_mime_types = excluded.allowed_mime_types;

-- ========== HELPER ==========
-- First path segment as a uuid, or null if it isn't one (never throws).
create function storage_folder_uuid(p_name text)
returns uuid language plpgsql immutable as $$
begin
  return ((storage.foldername(p_name))[1])::uuid;
exception when others then
  return null;
end $$;

######## FILE: 002_rls_tests.sql ########
-- RLS / privacy tests for 001_mvp_schema.sql
-- Prerequisites: 001_mvp_schema.sql and 004_join_rate_limit.sql applied.
-- Run in the Supabase SQL editor against a DEV project (never production).
-- Everything runs in one transaction and is rolled back at the end.
-- Success = a NOTICE saying "ALL RLS TESTS PASSED". Any failure raises an error.
-- If the auth.users insert fails on your Supabase version, add the NOT NULL
-- columns it names (they vary between versions).

begin;

-- Helper: act as a given user, the way an authenticated API request would.
create function pg_temp.as_user(u uuid) returns void language plpgsql as $$
begin
  execute 'reset role';
  perform set_config('request.jwt.claim.sub', u::text, true);
  perform set_config('request.jwt.claims',
    json_build_object('sub', u::text, 'role', 'authenticated')::text, true);
  execute 'set local role authenticated';
end $$;

do $$
declare

type Coords = { latitude: number; longitude: number };

// Great-circle distance in meters, used only for the client-side movement
// threshold. Distances shown to users come from PostGIS via the RPC.
export function haversineM(a: Coords, b: Coords): number {
  const R = 6371008.8;
  const rad = (d: number) => (d * Math.PI) / 180;
  const dLat = rad(b.latitude - a.latitude);
  const dLon = rad(b.longitude - a.longitude);
  const h =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(rad(a.latitude)) * Math.cos(rad(b.latitude)) * Math.sin(dLon / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(h));
}

export function formatDistance(m: number | null | undefined): string {
  if (m == null) return 'Distance unknown';
  return m < 1000 ? `${Math.round(m)} m` : `${(m / 1000).toFixed(1)} km`;
}

export type Freshness = { state: 'live' | 'recent' | 'stale'; label: string };

// "Live" only when the SERVER received an update in the last 2 minutes.
export function freshness(updatedAt: string, now = Date.now()): Freshness {
  const s = Math.max(0, Math.round((now - new Date(updatedAt).getTime()) / 1000));
  if (s < 120) return { state: 'live', label: s < 10 ? 'Live' : `Live · ${s}s ago` };
  const m = Math.round(s / 60);
  if (m <= 15) return { state: 'recent', label: `Updated ${m} min ago` };
  return {
    state: 'stale',
    label: m < 60 ? `Last seen ${m} min ago` : `Last seen ${Math.round(m / 60)} h ago`,
  };
}

import * as Location from 'expo-location';
import { supabase } from './supabase';
import { haversineM } from './geo';

// Starting values from the plan. Tune them on real devices.
const MIN_MOVE_M = 25;
const HEARTBEAT_MS = 60_000;
const MAX_ACCURACY_M = 100;
const MAX_FIX_AGE_MS = 30_000;

export type ShareStatus = 'starting' | 'active' | 'denied' | 'error';

/**
 * Foreground-only sharing. Returns a stop() function.
 * Not yet implemented: low-battery backoff, background tracking.
 */
export async function startSharing(
  userId: string,
  onStatus: (s: ShareStatus) => void
): Promise<() => void> {
  onStatus('starting');

  const perm = await Location.requestForegroundPermissionsAsync();
  if (perm.status !== 'granted') {
    onStatus('denied');
    return () => {};
  }

  let last: { latitude: number; longitude: number } | null = null;
  let lastSentAt = 0;
  let failures = 0;
  let stopped = false;

  async function onFix(f: Location.LocationObject) {
    if (stopped) return;
    const acc = f.coords.accuracy ?? 999;

    if (Date.now() - f.timestamp > MAX_FIX_AGE_MS) return;   // stale fix
    if (last && acc > MAX_ACCURACY_M) return;                // too imprecise

    const moved = last ? haversineM(last, f.coords) : Infinity;
    const due = Date.now() - lastSentAt > HEARTBEAT_MS;
    if (moved < Math.max(MIN_MOVE_M, acc) && !due) return;   // jitter / no news

    // updated_at is set by the database, not sent from the phone.
    const { error } = await supabase.from('user_locations').upsert({
      user_id: userId,
      geog: `SRID=4326;POINT(${f.coords.longitude} ${f.coords.latitude})`,
      accuracy_m: acc,
    });

    if (error) {
      failures += 1;
      if (failures >= 3) onStatus('error');
      return;  // the next fix or heartbeat retries; we don't queue a backlog
    }
    failures = 0;
    last = { latitude: f.coords.latitude, longitude: f.coords.longitude };
    lastSentAt = Date.now();
    onStatus('active');
  }

  // First fix immediately, so others see you as soon as sharing starts.
  try {
    await onFix(await Location.getCurrentPositionAsync({ accuracy: Location.Accuracy.Balanced }));
  } catch {
    onStatus('error');
  }

  const sub = await Location.watchPositionAsync(
    { accuracy: Location.Accuracy.Balanced, distanceInterval: 10, timeInterval: 5000 },
    onFix
  );

  // iOS only fires the watcher on movement, so a timer takes a fresh reading
  // when you're stationary. It reads real GPS, it doesn't resend an old point.
  const timer = setInterval(async () => {
    try {
      await onFix(await Location.getCurrentPositionAsync({ accuracy: Location.Accuracy.Balanced }));
    } catch {
      onStatus('error');
    }
  }, HEARTBEAT_MS);

  return () => {
    stopped = true;
    sub.remove();
    clearInterval(timer);
  };
}

  async function submit() {
    if (!email || !password) return Alert.alert('Enter your email and password.');
    if (mode === 'signup') {
      if (!/^[a-z0-9_]{3,24}$/.test(username))

      if (!fullName.trim()) return Alert.alert('Enter your name.');
      if (password.length < 8) return Alert.alert('Password must be at least 8 characters.');

    setBusy(true);
    try {
      if (mode === 'signin') {
        const { error } = await supabase.auth.signInWithPassword({ email: email.trim(), password });

      } else {

          password,
          options: { data: { username, full_name: fullName.trim() } },

        if (error) throw error;

        if (!data.session) Alert.alert('Check your email', 'Confirm your address, then sign in.');

      setBusy(false);

    if (error && error.status !== 400) return Alert.alert('Could not send the email', error.message);
    Alert.alert('If that account exists, a reset email is on its way.');
    // TODO week 2: deep link into an in-app "set new password" screen.
  }

  return (
    <KeyboardAvoidingView
      style={styles.wrap}
      behavior={Platform.OS === 'ios' ? 'padding' : undefined}
    >
      <Text style={styles.title}>{mode === 'signin' ? 'Sign in' : 'Create account'}</Text>

      {mode === 'signup' && (

            onChangeText={setFullName} accessibilityLabel="Full name" />
          <TextInput style={styles.input} placeholder="Username" autoCapitalize="none"
            value={username} onChangeText={(t) => setUsername(t.toLowerCase())}

      )}

        keyboardType="email-address" autoComplete="email" value={email}
        onChangeText={setEmail} accessibilityLabel="Email" />
      <TextInput style={styles.input} placeholder="Password" secureTextEntry
        autoComplete={mode === 'signin' ? 'current-password' : 'new-password'}

    setBusy(false);
    if (error) return Alert.alert('Could not join', 'Invalid room ID or password.');
    if (data === 'rate_limited') return Alert.alert('Too many attempts', 'Try again in 15 minutes.');
    if (data !== 'ok') return Alert.alert('Could not join', 'Invalid room ID or password.');
    setPanel('none'); setRoomId(''); setPassword('');
    load();
  }

  return (
    <View style={styles.wrap}>
      <View style={styles.header}>
        <Text style={styles.title}>Rooms</Text>
        <Pressable onPress={() => supabase.auth.signOut()}>
          <Text style={styles.link}>Sign out</Text>
        </Pressable>
      </View>

      <View style={styles.row}>
        <Pressable style={styles.btn} onPress={() => setPanel(panel === 'create' ? 'none' : 'create')}>
          <Text style={styles.btnText}>Create room</Text>
        </Pressable>
        <Pressable style={styles.btn} onPress={() => setPanel(panel === 'join' ? 'none' : 'join')}>
          <Text style={styles.btnText}>Join room</Text>
        </Pressable>
      </View>

      {panel !== 'none' && (
        <View style={styles.panel}>
          {panel === 'create' ? (
            <TextInput style={styles.input} placeholder="Room name" value={name} onChangeText={setName} />
          ) : (
            <TextInput style={styles.input} placeholder="Room ID" autoCapitalize="none"
              value={roomId} onChangeText={setRoomId} />
          )}
          <TextInput style={styles.input} placeholder="Password" secureTextEntry
            value={password} onChangeText={setPassword} />
          <Pressable style={[styles.btn, busy && { opacity: 0.6 }]} disabled={busy}
            onPress={panel === 'create' ? createRoom : joinRoom}>
            {busy ? <ActivityIndicator color="#fff" /> :
              <Text style={styles.btnText}>{panel === 'create' ? 'Create' : 'Join'}</Text>}
          </Pressable>
        </View>
      )}

      {loading ? <ActivityIndicator style={{ marginTop: 24 }} /> :
        error ? (
          <View style={styles.center}>
            <Text>Couldn't load your rooms.</Text>
            <Pressable onPress={load}><Text style={styles.link}>Retry</Text></Pressable>
          </View>
        ) : (
          <FlatList
            data={rooms}
            keyExtractor={(r) => r.id}
            ListEmptyComponent={
              <Text style={styles.empty}>No rooms yet. Create one, or join with a Room ID and password.</Text>
            }
            renderItem={({ item }) => (
              <Pressable style={styles.card}
                onPress={() => router.push({ pathname: '/(tabs)/map', params: { roomId: item.id, roomName: item.name } })}>
                <Text style={styles.cardTitle}>{item.name}</Text>
                <Text style={styles.cardSub}>{item.role}</Text>
              </Pressable>
            )}
          />
        )}
    </View>
  );
}

import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import { Alert, Linking, Pressable, StyleSheet, Text, View } from 'react-native';
import { useLocalSearchParams } from 'expo-router';
import MapSurface, { MapMember } from '../../components/MapSurface';
import { supabase } from '../../lib/supabase';
import { formatDistance, freshness } from '../../lib/geo';
import { ShareStatus, startSharing } from '../../lib/locationSharing';

type LocRow = {
  user_id: string; lat: number; lng: number;
  accuracy_m: number | null; updated_at: string; distance_m: number | null;
};

const SHARE_HOURS = 1;  // safe default: sharing expires on its own

export default function MapScreen() {
  const { roomId, roomName } = useLocalSearchParams<{ roomId?: string; roomName?: string }>();
  const [userId, setUserId] = useState<string | null>(null);
  const [rows, setRows] = useState<LocRow[]>([]);
  const [names, setNames] = useState<Record<string, string>>({});
  const [loadError, setLoadError] = useState(false);
  const [sharing, setSharing] = useState(false);
  const [expiresAt, setExpiresAt] = useState<string | null>(null);
  const [shareStatus, setShareStatus] = useState<ShareStatus | null>(null);
  const [mapType, setMapType] = useState<'standard' | 'satellite' | 'hybrid'>('standard');
  const [now, setNow] = useState(Date.now());
  const reloadTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    supabase.auth.getUser().then(({ data }) => setUserId(data.user?.id ?? null));
  }, []);

  // Refresh "Live / Updated X ago" labels every 15 s.
  useEffect(() => {
    const t = setInterval(() => setNow(Date.now()), 15_000);
    return () => clearInterval(t);
  }, []);

  const load = useCallback(async () => {
    if (!roomId || !userId) return;
    const { data, error } = await supabase.rpc('room_member_locations', { p_room: roomId });
    if (error) { setLoadError(true); return; }
    setLoadError(false);
    const list = (data ?? []) as LocRow[];
    setRows(list);

    const ids = list.map((r) => r.user_id);
    if (ids.length) {
      const { data: profs } = await supabase.from('profiles').select('id, full_name').in('id', ids);
      setNames(Object.fromEntries((profs ?? []).map((p: any) => [p.id, p.full_name])));
    }

    const { data: mine } = await supabase
      .from('room_members').select('sharing_enabled, share_expires_at')
      .eq('room_id', roomId).eq('user_id', userId).maybeSingle();
    const active = !!mine?.sharing_enabled &&
      (!mine.share_expires_at || new Date(mine.share_expires_at).getTime() > Date.now());
    setSharing(active);
    setExpiresAt(active ? mine?.share_expires_at ?? null : null);
  }, [roomId, userId]);

  // Initial load + realtime. Events only signal "something changed"; we re-fetch
  // through the RPC so RLS and server-side distance still apply.
  useEffect(() => {
    load();
    if (!roomId) return;
    const refresh = () => {
      if (reloadTimer.current) clearTimeout(reloadTimer.current);
      reloadTimer.current = setTimeout(load, 500);
    };
    const channel = supabase.channel(`room-${roomId}`)
      .on('postgres_changes', { event: '*', schema: 'public', table: 'user_locations' }, refresh)
      .on('postgres_changes', { event: '*', schema: 'public', table: 'room_members', filter: `room_id=eq.${roomId}` }, refresh)
      .subscribe();
    return () => { supabase.removeChannel(channel); };
  }, [load, roomId]);

  // Run the location loop only while sharing is on for this room.
  useEffect(() => {
    if (!sharing || !userId) { setShareStatus(null); return; }
    let stop: (() => void) | null = null;
    let cancelled = false;
    startSharing(userId, setShareStatus).then((s) => {
      if (cancelled) s(); else stop = s;
    });
    // Stop at expiry, then reload so the UI reflects it.
    const ms = expiresAt ? new Date(expiresAt).getTime() - Date.now() : null;
    const expiry = ms != null ? setTimeout(load, Math.max(ms, 0) + 500) : null;
    return () => { cancelled = true; stop?.(); if (expiry) clearTimeout(expiry); };
  }, [sharing, userId, expiresAt, load]);

  async function toggleSharing() {
    if (!roomId || !userId) return;
    const turnOn = !sharing;
    const { error } = await supabase.from('room_members')
      .update({
        sharing_enabled: turnOn,
        share_expires_at: turnOn ? new Date(Date.now() + SHARE_HOURS * 3600_000).toISOString() : null,
      })
      .eq('room_id', roomId).eq('user_id', userId);
    if (error) return Alert.alert('Could not change sharing', error.message);
    load();
  }

  const members: MapMember[] = useMemo(() => rows.map((r) => {
    const f = freshness(r.updated_at, now);
    return {
      id: r.user_id,
      name: names[r.user_id] ?? 'Member',
      lat: r.lat, lng: r.lng,
      distanceText: formatDistance(r.distance_m),
      freshnessLabel: f.label,
      stale: f.state === 'stale',
      accuracyText: r.accuracy_m != null ? `Accuracy ±${Math.round(r.accuracy_m)} m` : 'Accuracy unknown',
    };
  }), [rows, names, now]);

  if (!roomId) {
    return <View style={styles.center}><Text>Pick a room on the Rooms tab to see its map.</Text></View>;
  }

  const banner =
    shareStatus === 'denied' ? 'Location permission is off. Enable it in Settings to share.' :
    shareStatus === 'error' ? "Can't reach the server or GPS. Your location may not be updating." :
    loadError ? "Can't load the latest locations. Check your connection." : null;

  return (
    <View style={styles.wrap}>
      <MapSurface members={members} mapType={mapType} />

      {banner && (
        <Pressable style={styles.banner} onPress={() => shareStatus === 'denied' ? Linking.openSettings() : load()}>
          <Text style={styles.bannerText}>{banner}</Text>
        </Pressable>
      )}

      <View style={styles.sheet}>
        <Text style={styles.room}>{roomName ?? 'Room'}</Text>
        <Text style={styles.sub}>
          {members.length === 0
            ? 'No one else is sharing their location right now.'
            : `${members.length} member${members.length === 1 ? '' : 's'} sharing`}
        </Text>
        <View style={styles.row}>
          <Pressable style={[styles.btn, sharing && styles.btnOn]} onPress={toggleSharing}>
            <Text style={styles.btnText}>
              {sharing ? `Sharing${shareStatus === 'active' ? ' · live' : '…'} (tap to stop)` : `Share my location (${SHARE_HOURS} h)`}
            </Text>
          </Pressable>
          <Pressable style={styles.btnSmall}
            onPress={() => setMapType(mapType === 'standard' ? 'hybrid' : mapType === 'hybrid' ? 'satellite' : 'standard')}>
            <Text style={styles.btnText}>{mapType}</Text>
          </Pressable>
        </View>
        <Text style={styles.note}>Sharing works while the app is open. It stops on its own after {SHARE_HOURS} hour.</Text>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  wrap: { flex: 1 },
  center: { flex: 1, alignItems: 'center', justifyContent: 'center', padding: 24 },
  banner: { position: 'absolute', top: 56, left: 16, right: 16, backgroundColor: '#b91c1c', padding: 12, borderRadius: 10 },
  bannerText: { color: '#fff', fontWeight: '600' },
  sheet: { position: 'absolute', bottom: 0, left: 0, right: 0, backgroundColor: '#fff', padding: 16, paddingBottom: 28, borderTopLeftRadius: 18, borderTopRightRadius: 18, gap: 6 },
  room: { fontSize: 18, fontWeight: '700' },
  sub: { color: '#4b5563' },
  row: { flexDirection: 'row', gap: 10, marginTop: 6 },
  btn: { flex: 1, backgroundColor: '#2563eb', padding: 14, borderRadius: 10, alignItems: 'center' },
  btnOn: { backgroundColor: '#15803d' },
  btnSmall: { backgroundColor: '#374151', paddingHorizontal: 14, justifyContent: 'center', borderRadius: 10 },
  btnText: { color: '#fff', fontWeight: '600' },
  note: { color: '#6b7280', fontSize: 12 },
});
